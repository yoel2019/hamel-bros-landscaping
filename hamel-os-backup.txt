const EmployeePanel = ({ onLogout, appData, setAppData, isDemoMode }) => {
  const [pin, setPin] = useState('');
  const [err, setErr] = useState(false);
  const [currentTime, setCurrentTime] = useState(nowTime());
  useEffect(()=>{ const iv=setInterval(()=>setCurrentTime(nowTime()),10000); return ()=>clearInterval(iv); },[]);

  const handleKey = n => { if(pin.length<4){setErr(false);setPin(pin+n);} };

  const handleSubmit = () => {
    if(pin.length!==4) return;
    const emp=appData.employees.find(e=>e.pin===pin);
    if(!emp){setErr(true);setPin('');return;}
    setAppData(prev=>{
      const next=JSON.parse(JSON.stringify(prev));
      const e=next.employees.find(x=>x.id===emp.id);
      const type=e.clockedIn?'out':'in';
      e.clockedIn=!e.clockedIn;
      const t=nowTime();
      e.punchLog.unshift({type,time:t});
      next.punches.unshift({type,time:t,name:e.name});
      next.isClockedIn=e.clockedIn;
      return next;
    });
    setPin('');
  };

  return (
    <div className={`w-full h-full bg-brand-black flex flex-col ${!isDemoMode&&'min-h-screen sm:h-[850px] max-w-md mx-auto sm:border-8 sm:border-brand-border sm:rounded-[2.5rem]'}`}>

        <h2 className="text-base font-bold text-white flex items-center gap-2">
          {!isDemoMode&&<button onClick={onLogout} className="text-brand-muted hover:text-white"><Ico.Logout/></button>}

        <BrandLogo h="h-7"/>
      </header>
      <div className="flex-1 flex flex-col items-center p-5 justify-center">

const App = () => {
  const [appData, setAppData] = useState(initialData);
  const [role, setRole] = useState(null);

  // GPS simulation
  useEffect(()=>{
    let iv;
    const hasEnRoute = appData.crews.some(c=>c.status==='en_route');
    if(hasEnRoute){
      iv=setInterval(()=>{
        setAppData(prev=>{
          const next=JSON.parse(JSON.stringify(prev));
          next.crews.forEach(c=>{
            if(c.status==='en_route'&&c.gpsProgress<1){
              c.gpsProgress=Math.min(1,c.gpsProgress+0.04);
              c.liveSpeed=Math.floor(Math.random()*15)+28;
              c.coords.x+=(65-c.coords.x)*0.04;
              c.coords.y+=(48-c.coords.y)*0.04;
            }
          });
          return next;
        });
      },1200);
    }
    return ()=>clearInterval(iv);
  },[appData.crews.map(c=>c.status).join()]);

  return (
    <React.Fragment>
      {!role         && <LoginScreen onSelectRole={setRole}/>}
      {role==='demo' && <DemoSplitScreen onLogout={()=>setRole(null)} appData={appData} setAppData={setAppData}/>}
      {role==='admin'    && <AdminPanel    onLogout={()=>setRole(null)} appData={appData} setAppData={setAppData} isDemoMode={false}/>}
      {role==='leader'   && <LeaderPanel   onLogout={()=>setRole(null)} appData={appData} setAppData={setAppData} isDemoMode={false}/>}
      {role==='employee' && <EmployeePanel onLogout={()=>setRole(null)} appData={appData} setAppData={setAppData} isDemoMode={false}/>}
      {role==='client'   && <ClientTracker onLogout={()=>setRole(null)} appData={appData} isDemoMode={false}/>}
    </React.Fragment>
  );
};

ReactDOM.createRoot(document.getElementById('root')).render(<App/>);
</script>
